\section{The infinite spine case}

Throughout this section, $G$ denotes an ordered abelian group.
For $n \geq 2$ and $a \notin nG$, recall that $\mathfrak{s}_n(a)$ is the largest convex subgroup $H \Subset G$ such that $a \notin H + nG$.
If $a \in nG$, we set $\mathfrak{s}_n(a) = \emptyset$, and we regard $\emptyset$ as being smaller than every convex subgroup.
We set $\mathcal{S}_n = \left\{ \mathfrak{s}_n(a) \mid a \in G \right\}$.
The subgroups $\mathfrak{s}_n(a)$ are uniformly definable in $a$ \cite[Lemma 2.1]{Cluckers2011QuaEli}.
Consequently, comparisons and equalities between values of the maps $\mathfrak{s}_n$ are formulas in $L_{\mathrm{oag}}$.

\begin{lem}[{\cite[Lemma 2.8(2) and Remark 2.6]{Okura2026DisOrd}}]\label{lem:BasicPropertiesOfS}
	Let $p$ be a prime, $r \geq 1$, and $a,b \in G$.
	\begin{enumerate}
		\item If $0 \leq j < r$, then
		\[
			\mathfrak{s}_{p^r}(p^j a) = \mathfrak{s}_{p^{r-j}}(a).
		\]
		If $j \geq r$, then $\mathfrak{s}_{p^r}(p^j a) = \emptyset$.
		\item If $\mathfrak{s}_{p^r}(a) \neq \mathfrak{s}_{p^r}(b)$, then
		\[
			\mathfrak{s}_{p^r}(a+b)
			=
			\max \left\{ \mathfrak{s}_{p^r}(a),\mathfrak{s}_{p^r}(b) \right\}.
		\]
	\end{enumerate}
\end{lem}

\begin{lem}\label{lem:RepresentativeBetweenSValues}
	Let $p$ be a prime and let $H<H'$ be nonempty elements of $\mathcal{S}_p$.
	There exists $a\in H'$ such that
	\[
		\mathfrak{s}_p(a)=H
		\quad\text{and}\quad
		\mathfrak{s}_{p^r}(a)<H'
	\]
	for every $r\geq 1$.
\end{lem}

\begin{proof}
	Choose $b\in G$ such that $\mathfrak{s}_p(b)=H$.
	Since $H'>H$, the maximality of $H$ gives $b\in H'+pG$.
	We may therefore write $b=a+pc$ with $a\in H'$ and $c\in G$.
	As $a-b\in pG$, we have $\mathfrak{s}_p(a)=\mathfrak{s}_p(b)=H$.

	Fix $r\geq 1$.
	If $\mathfrak{s}_{p^r}(a)=\emptyset$, there is nothing to prove.
	Otherwise, put $K=\mathfrak{s}_{p^r}(a)$.
	If $K\geq H'$, then $a\in H'\subseteq K$, contrary to $a\notin K+p^rG$.
	Hence, $K<H'$.
\end{proof}

\begin{lem}[Finite coding]\label{lem:FiniteCoding}
	Let $p$ be a prime and
	\[
		K_0<K_1<\dots<K_N
	\]
	be nonempty elements of $\mathcal{S}_p$.
	There exists $x\in G$ such that
	\[
		\mathfrak{s}_{p^{i+1}}(x)=K_i
	\]
	for every $0\leq i\leq N$.
\end{lem}

\begin{proof}
	For each $j<N$, use \autoref{lem:RepresentativeBetweenSValues} to choose $a_j\in G$ such that
	\[
		\mathfrak{s}_p(a_j)=K_j
		\quad\text{and}\quad
		\mathfrak{s}_{p^r}(a_j)<K_{j+1}
	\]
	for every $r\geq 1$.
	Also choose $a_N\in G$ with $\mathfrak{s}_p(a_N)=K_N$, and set
	\[
		x=a_0+pa_1+\dots+p^Na_N.
	\]
	Fix $0\leq i\leq N$.
	By \autoref{lem:BasicPropertiesOfS},
	\[
		\mathfrak{s}_{p^{i+1}}(p^ia_i)=\mathfrak{s}_p(a_i)=K_i.
	\]
	If $j<i$, then
	\[
		\mathfrak{s}_{p^{i+1}}(p^ja_j)
		=
		\mathfrak{s}_{p^{i+1-j}}(a_j)
		<
		K_{j+1}
		\leq
		K_i,
	\]
	while, if $j>i$, then $\mathfrak{s}_{p^{i+1}}(p^ja_j)=\emptyset$.
	Thus, among the summands of $x$, the unique maximal $\mathfrak{s}_{p^{i+1}}$-value is $K_i$.
	Repeatedly applying \autoref{lem:BasicPropertiesOfS}(2) gives $\mathfrak{s}_{p^{i+1}}(x)=K_i$.
\end{proof}

\begin{thm}\label{thm:InfiniteSpine}
	If $\mathcal{S}_p$ is infinite for some prime $p$, then the dp-rank of $G$ is $\aleph_0$.
\end{thm}

\begin{proof}
	We work in a sufficiently saturated monster model of $\operatorname{Th}(G)$.
	Since $\mathcal{S}_p$ is an infinite interpretable linear order, compactness allows us to choose elements
	\[
		(b_{i,k})_{i,k<\omega}
	\]
	such that all the values
	\[
		H_{i,k}:=\mathfrak{s}_p(b_{i,k})
	\]
	are nonempty and
	\[
		H_{i,k}<H_{j,l}
		\quad\text{whenever}\quad
		i<j
		\ \text{or}\
		(i=j\ \text{and}\ k<l).
	\]
	Indeed, every finite part of these requirements merely asks for a finite chain of distinct nonempty elements of the infinite linear order $\mathcal{S}_p$.

	For a function $\eta:\omega\to\omega$, consider the partial type
	\[
		\Sigma_\eta(x)
		=
		\left\{
			\mathfrak{s}_{p^{i+1}}(x)=H_{i,\eta(i)}
			\;\middle|\;
			i<\omega
		\right\}.
	\]
	This type is finitely satisfiable.
	Indeed, given $N<\omega$, the values
	\[
		H_{0,\eta(0)}<H_{1,\eta(1)}<\dots<H_{N,\eta(N)}
	\]
	form a finite increasing chain, so \autoref{lem:FiniteCoding} realizes the first $N+1$ conditions of $\Sigma_\eta$.
	By saturation, there is $a_\eta\in G$ realizing $\Sigma_\eta$.

	For every $i<\omega$, let
	\[
		\varphi_i(x;y)
		\quad\text{be the formula}\quad
		\mathfrak{s}_{p^{i+1}}(x)=\mathfrak{s}_p(y).
	\]
	As noted above, this is an $L_{\mathrm{oag}}$-formula.
	For all $\eta:\omega\to\omega$, $i<\omega$, and $k<\omega$, we have
	\[
		\models\varphi_i(a_\eta;b_{i,k})
		\quad\Longleftrightarrow\quad
		H_{i,\eta(i)}=H_{i,k}
		\quad\Longleftrightarrow\quad
		\eta(i)=k.
	\]
	Hence, these formulas and parameters form an ICT-pattern of depth $\aleph_0$, and therefore the dp-rank of $G$ is at least $\aleph_0$.

	Every ordered abelian group has NIP \cite{Gurevich1984TheOrd}.
	Since $L_{\mathrm{oag}}$ is countable, this gives dp-rank at most $\aleph_0$.
	Consequently, the dp-rank of $G$ is exactly $\aleph_0$.
\end{proof}

\section{The lower bound in the finite-spine case}

We now assume that $G$ is nontrivial and has finite spines.
For a prime $p$ and $H\in\mathcal{S}_p\setminus\{\emptyset\}$, recall that
\[
	H^{[p]}=\bigcap_{K\Subset G,\,K>H}(K+pG)
\]
and
\[
	R_H^{\mathfrak{s}_p}
	=
	H^{[p]}/(H+pG)
\]
by \cite[Definition 2.10 and Lemma 2.14(1)]{Okura2026DisOrd}.
We also use $\mathcal{S}_{p^r}=\mathcal{S}_p$ for every $r\geq 1$, as established in \cite[Lemma 2.9(3)]{Okura2026DisOrd}.
We put
\[
	\mathcal{S}_p^\infty
	=
	\left\{
		H\in\mathcal{S}_p\setminus\{\emptyset\}
		\;\middle|\;
		\left|R_H^{\mathfrak{s}_p}\right|=\infty
	\right\}.
\]

\begin{lem}\label{lem:RepresentativesOfInfiniteRib}
	Let $p$ be a prime and $H\in\mathcal{S}_p^\infty$.
	There is a sequence $(a^k)_{k<\omega}$ such that
	\[
		\mathfrak{s}_p(a^k)
		=
		\mathfrak{s}_p(a^k-a^l)
		=
		H
	\]
	whenever $k\neq l$.
	If $H<K$ for some $K\in\mathcal{S}_p\setminus\{\emptyset\}$, the sequence may moreover be chosen inside $K$.
\end{lem}

\begin{proof}
	Choose elements $(u^k)_{k<\omega}$ whose images in
	\[
		R_H^{\mathfrak{s}_p}=H^{[p]}/(H+pG)
	\]
	are nonzero and pairwise distinct.
	Thus, neither $u^k$ nor $u^k-u^l$ belongs to $H+pG$, while all these elements belong to $H^{[p]}$.
	By \cite[Lemma 2.14(1)]{Okura2026DisOrd}, this says precisely that
	\[
		\mathfrak{s}_p(u^k)
		=
		\mathfrak{s}_p(u^k-u^l)
		=
		H
	\]
	for $k\neq l$.

	Suppose that $H<K$.
	Since $u^k\in H^{[p]}\subseteq K+pG$, we can write $u^k=a^k+pc_k$ with $a^k\in K$.
	Replacing $u^k$ by $a^k$ does not change its image in $R_H^{\mathfrak{s}_p}$.
	The images therefore remain nonzero and pairwise distinct, so the required equalities remain true.
\end{proof}

\begin{lem}[Simultaneous coding for one prime]\label{lem:SimultaneousCodingOnePrime}
	Let $p$ be a prime and let
	\[
		H_0<H_1<\dots<H_{n-1}
	\]
	be elements of $\mathcal{S}_p^\infty$.
	For each $i<n$, there is a sequence $(a_i^k)_{k<\omega}$ as in \autoref{lem:RepresentativesOfInfiniteRib}, and these sequences can be chosen so that $a_i^k\in H_{i+1}$ whenever $i<n-1$.
	For $\zeta:n\to\omega$, set
	\[
		b_\zeta
		=
		a_0^{\zeta(0)}
		+pa_1^{\zeta(1)}
		+\dots
		+p^{n-1}a_{n-1}^{\zeta(n-1)}.
	\]
	Then, for every $i<n$,
	\begin{enumerate}
		\item $\mathfrak{s}_{p^{i+1}}(b_\zeta)=H_i$;
		\item the image of $b_\zeta$ in $R_{H_i}^{\mathfrak{s}_{p^{i+1}}}$ is the same as that of $p^ia_i^{\zeta(i)}$;
		\item for every $k<\omega$,
		\[
			b_\zeta-p^ia_i^k\in H_i+p^{i+1}G
			\quad\Longleftrightarrow\quad
			k=\zeta(i).
		\]
	\end{enumerate}
\end{lem}

\begin{proof}
	The required sequences are supplied by \autoref{lem:RepresentativesOfInfiniteRib}.
	For $j<n-1$, the fact that $a_j^k\in H_{j+1}$ implies
	\[
		\mathfrak{s}_{p^r}(a_j^k)<H_{j+1}
	\]
	for every $r\geq 1$: otherwise, $\mathfrak{s}_{p^r}(a_j^k)\geq H_{j+1}$ would contain $a_j^k$, contrary to the definition of $\mathfrak{s}_{p^r}$.
	The calculation in the proof of \autoref{lem:FiniteCoding} now gives
	\[
		\mathfrak{s}_{p^{i+1}}(b_\zeta)=H_i,
	\]
	which proves (1).

	For $j<i$, we have $a_j^{\zeta(j)}\in H_{j+1}\subseteq H_i$, whereas for $j>i$, the term $p^ja_j^{\zeta(j)}$ belongs to $p^{i+1}G$.
	Hence,
	\[
		b_\zeta-p^ia_i^{\zeta(i)}
		\in
		H_i+p^{i+1}G.
	\]
	Also, by \autoref{lem:BasicPropertiesOfS},
	\[
		\mathfrak{s}_{p^{i+1}}(p^ia_i^{\zeta(i)})
		=
		\mathfrak{s}_p(a_i^{\zeta(i)})
		=
		H_i.
	\]
	Thus, both elements represent classes in
	\[
		R_{H_i}^{\mathfrak{s}_{p^{i+1}}}
		=
		H_i^{[p^{i+1}]}/(H_i+p^{i+1}G),
	\]
	and the congruence $b_\zeta - p^i a_i^{\zeta(i)} \in H_i + p^{i+1}G$ proves (2).

	If $k\neq\zeta(i)$, then
	\[
		p^i\left(a_i^{\zeta(i)}-a_i^k\right)
		\notin
		H_i+p^{i+1}G.
	\]
	Indeed, by the purity of the convex subgroup $H_i$, membership in the group on the right would imply
	\[
		a_i^{\zeta(i)}-a_i^k\in H_i+pG,
	\]
	contrary to \autoref{lem:RepresentativesOfInfiniteRib}.
	Together with the preceding congruence, this proves (3).
\end{proof}

The following standard consequences of compactness will be used without proof.
For the second assertion, finite satisfiability follows from the Chinese remainder theorem \cite[Lemma 2.1(2)]{Okura2026DisOrd}.

\begin{lem}\label{lem:CofinalDivisibleTranslations}
	Let $G$ be a sufficiently saturated monster model of the theory of a nontrivial ordered abelian group.
	\begin{enumerate}
		\item The type-definable subgroup
		\[
			D=\bigcap_{m\geq 2}mG
		\]
		is cofinal and coinitial in $G$.
		\item Let $P_0\subseteq\mathbb{P}$ and $(a_p)_{p\in P_0}$ be a sequence from $G$.
		There exists $a\in G$ such that
		\[
			a-a_p\in p^rG
		\]
		for every $p\in P_0$ and $r\geq 1$.
	\end{enumerate}
\end{lem}

\begin{thm}\label{thm:LowerBoundFiniteSpines}
	Let $G$ be a nontrivial ordered abelian group with finite spines.
	Then
	\[
		\operatorname{dp\text{-}rk}(G)
		\geq
		1+\sum_{p\in\mathbb{P}}\left|\mathcal{S}_p^\infty\right|.
	\]
	If the sum on the right is infinite, then the dp-rank of $G$ is $\aleph_0$.
\end{thm}

\begin{proof}
	We work in a sufficiently saturated monster model.
	For each prime $p$, put $n_p=\left|\mathcal{S}_p^\infty\right|$ and, when $n_p>0$, enumerate
	\[
		\mathcal{S}_p^\infty
		=
		\{H_{p,0}<\dots<H_{p,n_p-1}\}.
	\]
	The number $n_p$ is finite because $G$ has finite spines.
	Let
	\[
		\mathcal{I}
		=
		\{(p,i)\mid p\in\mathbb{P},\ 0\leq i<n_p\}.
	\]
	Thus,
	\[
		|\mathcal{I}|
		=
		\sum_{p\in\mathbb{P}}\left|\mathcal{S}_p^\infty\right|,
	\]
	and $\mathcal{I}$ is finite or countable.

	For each $p$ with $n_p>0$, choose sequences $(a_{p,i}^k)_{k<\omega}$ as in \autoref{lem:SimultaneousCodingOnePrime}, and denote the corresponding coding elements by $b_{p,\zeta}$ for $\zeta:n_p\to\omega$.
	For $(p,i)\in\mathcal{I}$, let
	\[
		\varphi_{p,i}(x;y)
		\quad\text{be the formula}\quad
		x-y\in\mathfrak{s}_{p^{i+1}}(y)+p^{i+1}G.
	\]
	This is an $L_{\mathrm{oag}}$-formula by the uniform definability of $\mathfrak{s}_{p^{i+1}}$.
	We use the parameter sequence
	\[
		\left(p^ia_{p,i}^k\right)_{k<\omega}
	\]
	in the row indexed by $(p,i)$.
	By \autoref{lem:BasicPropertiesOfS},
	\[
		\mathfrak{s}_{p^{i+1}}(p^ia_{p,i}^k)
		=
		H_{p,i}.
	\]

	Fix a function $\eta:\mathcal{I}\to\omega$ and consider
	\[
		q_\eta(x)
		=
		\left\{
			\varphi_{p,i}
			\left(x;p^ia_{p,i}^k\right)^{\eta(p,i)=k}
			\;\middle|\;
			(p,i)\in\mathcal{I},\ k<\omega
		\right\}.
	\]
	Here, the superscript means that the formula is negated when $\eta(p,i)\neq k$.
	For each $p$ with $n_p>0$, let
	\[
		\eta_p:n_p\to\omega,
		\qquad
		\eta_p(i)=\eta(p,i).
	\]
	By \autoref{lem:CofinalDivisibleTranslations}(2), there is $b_\eta\in G$ such that
	\[
		b_\eta-b_{p,\eta_p}\in p^rG
	\]
	for every $p$ with $n_p>0$ and $r\geq 1$.
	Part (3) of \autoref{lem:SimultaneousCodingOnePrime} then shows that $b_\eta\models q_\eta$.

	It remains to add the row arising from the order.
	Since the monster is sufficiently saturated, there exists $c>0$ such that
	\[
		-c<b_\eta<c
	\]
	for every $\eta:\mathcal{I}\to\omega$.
	By \autoref{lem:CofinalDivisibleTranslations}, choose
	\[
		t_0<t_1<\dots
	\]
	from $D$ so that $t_{k+1}-t_k>2c$ for every $k<\omega$.
	Set
	\[
		I_k=(t_k-c,t_k+c).
	\]
	These intervals are pairwise disjoint and satisfy $I_0<I_1<\dots$.
	We take
	\[
		\psi(x;y,z):=(y<x<z)
	\]
	for the additional row, with parameter sequence
	\[
		(t_k-c,t_k+c)_{k<\omega}.
	\]

	Let $\nu:\{\ast\}\mathbin{\dot\cup}\mathcal{I}\to\omega$, and put $\eta=\nu|_{\mathcal{I}}$.
	The element
	\[
		a_\nu=b_\eta+t_{\nu(\ast)}
	\]
	belongs to $I_{\nu(\ast)}$.
	Moreover, $t_{\nu(\ast)}\in p^{i+1}G$ for every $(p,i)\in\mathcal{I}$, so translation by $t_{\nu(\ast)}$ does not change the truth of any formula $\varphi_{p,i}$.
	It follows that
	\[
		\models\psi(a_\nu;t_k-c,t_k+c)
		\quad\Longleftrightarrow\quad
		\nu(\ast)=k
	\]
	and
	\[
		\models
		\varphi_{p,i}
		\left(a_\nu;p^ia_{p,i}^k\right)
		\quad\Longleftrightarrow\quad
		\nu(p,i)=k.
	\]
	We have therefore constructed an ICT-pattern of depth
	\[
		1+|\mathcal{I}|
		=
		1+\sum_{p\in\mathbb{P}}\left|\mathcal{S}_p^\infty\right|,
	\]
	as required.
	If the sum is infinite, the ICT-pattern has depth $\aleph_0$.
	The NIP upper bound recalled in the proof of \autoref{thm:InfiniteSpine} gives $\operatorname{dp\text{-}rk}(G)\leq\aleph_0$, so equality holds in this case.
\end{proof}

\section{The upper bound in the finite-spine case}

We continue to assume that $G$ is nontrivial and has finite spines, and now also assume that
\[
	N
	=
	\sum_{p\in\mathbb{P}}
	\left|\mathcal{S}_p^\infty\right|
\]
is finite.
We use the following standard criterion.

\begin{fct}[Indiscernible-sequence criterion; {\cite[Theorem 4.18(iii)]{Simon2015GuiNIP}}]\label{fct:DpRankCriterion}
	Let $T$ be a complete theory and let $n\geq 1$.
	Suppose that, whenever $I\models\mathrm{DLO}$, $(\overline{a}_i)_{i\in I}$ is an indiscernible sequence, and $b$ is a singleton, there is a decomposition
	\[
		I_0<J_0<I_1<\dots<J_{n-1}<I_n
	\]
	such that each $I_l$ has no endpoints, each $J_l$ has at most one element, and
	\[
		\operatorname{tp}(\overline{a}_i/b)
		=
		\operatorname{tp}(\overline{a}_j/b)
	\]
	whenever $i,j\in I_l$.
	Then $\operatorname{dp\text{-}rk}(T)\leq n$.
\end{fct}

We will also use the following specialization of relative quantifier elimination.

\begin{prop}[{\cite[Proposition 2.26]{Okura2026DisOrd}}]\label{prop:FiniteSpineElementaryMap}
	Let $G$ be an ordered abelian group with finite spines and let $f$ be a partial map from $G$ to $G$ whose domain is a subgroup.
	Suppose that $f$ is an ordered group embedding and, for every $a$ in its domain, every prime $p$, and every $r\geq 1$, the following hold:
	\begin{align*}
		\mathfrak{s}_{p^r}(f(a))&=\mathfrak{s}_{p^r}(a),&
		\mathfrak{t}_p(f(a))&=\mathfrak{t}_p(a),\\
		a=_{\bullet}k_{\bullet}
		&\Longleftrightarrow
		f(a)=_{\bullet}k_{\bullet},&
		a\equiv_{p^r,\bullet}k_{\bullet}
		&\Longleftrightarrow
		f(a)\equiv_{p^r,\bullet}k_{\bullet}.
	\end{align*}
	Here the third condition is required for every $k\in\mathbb{Z}\setminus\{0\}$, and the fourth for every $1\leq k<p^r$.
	Then $f$ is elementary.
\end{prop}

\begin{proof}
	We use the identity map on all the auxiliary sorts in the cited proposition.
	The reduction of the predicates $x\equiv_{m,\bullet}k_{\bullet}$ to prime-power moduli is given by \cite[Lemma 2.22]{Okura2026DisOrd}.
	It only remains to explain why its predicates $D_{p^r}^{[p^s]}$, where $s\geq r$, require no additional hypothesis.

	Indeed, in the finite-spine case,
	\[
		D_{p^r}^{[p^s]}(a)
		\quad\Longleftrightarrow\quad
		a\notin p^rG.
	\]
	The forward implication is immediate from the definition.
	Conversely, put $H=\mathfrak{s}_{p^r}(a)$.
	If $H$ has a successor $H'$ in $\mathcal{S}_p$, then
	\[
		H^{[p^s]}=H'+p^sG,
	\]
	while, if $H$ has no successor, then $H^{[p^s]}=G$.
	In either case, the definition of $\mathfrak{s}_{p^r}(a)$ gives
	\[
		a\in H^{[p^s]}+p^rG
		\quad\text{and}\quad
		a\notin H+p^rG.
	\]
	Thus $H$ witnesses $D_{p^r}^{[p^s]}(a)$.
	Since preservation of $\mathfrak{s}_{p^r}$ also preserves membership in $p^rG$, the result follows.
\end{proof}

\begin{lem}[Higher prime-power ribs]\label{lem:HigherPrimePowerRibs}
	Let $p$ be a prime, $H\in\mathcal{S}_p\setminus\{\emptyset\}$, and $r\geq 1$.
	If $R_H^{\mathfrak{s}_p}$ is finite, then $R_H^{\mathfrak{s}_{p^r}}$ is finite.
	Consequently,
	\[
		\left|R_H^{\mathfrak{s}_{p^r}}\right|=\infty
		\quad\Longrightarrow\quad
		H\in\mathcal{S}_p^\infty.
	\]
\end{lem}

\begin{proof}
	If $H$ has a successor $H'$ in the finite chain $\mathcal{S}_p$, set $K=H'$; otherwise set $K=G$.
	Using $\mathcal{S}_{p^r}=\mathcal{S}_p$, we obtain
	\[
		H^{[p^r]}=K+p^rG.
	\]
	Convex subgroups are pure, so
	\[
		K\cap(H+p^rG)=H+p^rK.
	\]
	Consequently,
	\[
		R_H^{\mathfrak{s}_{p^r}}
		\cong
		(K/H)/p^r(K/H),
		\qquad
		R_H^{\mathfrak{s}_p}
		\cong
		(K/H)/p(K/H).
	\]
	For any abelian group $A$, finiteness of $A/pA$ implies finiteness of $A/p^rA$, by applying successively the quotients in the filtration
	\[
		A\supseteq pA\supseteq\dots\supseteq p^rA.
	\]
	This proves the lemma.
\end{proof}

The next lemma isolates the single contribution coming from the order and from the ordered ribs associated with the maps $\mathfrak{t}_p$.

\begin{lem}[The common order cut]\label{lem:CommonOrderCut}
	Let $I\models\mathrm{DLO}$, let $(\overline{a}_i)_{i\in I}$ be indiscernible, and let $b\in G$.
	There is a decomposition
	\[
		I^-<J<I^+
	\]
	with $|J|\leq 1$ and with $I^-$ and $I^+$ having no endpoints, such that the following data are constant as $i$ varies in $I^-$, and also constant as $i$ varies in $I^+$:
	\begin{enumerate}
		\item the sign of $\overline{m}\cdot\overline{a}_i-mb$;
		\item $\mathfrak{t}_p(\overline{m}\cdot\overline{a}_i-mb)$;
		\item the truth value of
		\[
			\overline{m}\cdot\overline{a}_i-mb
			=_{\bullet}k_{\bullet}.
		\]
	\end{enumerate}
	Here the assertions hold simultaneously for all integer tuples $\overline{m}$, all nonzero integers $m$, all primes $p$, and all nonzero integers $k$.
\end{lem}

\begin{proof}
	Let $\mathcal{H}$ consist of $\{0\}$ and all nonempty convex subgroups occurring in one of the finite auxiliary sorts $\mathcal{S}_p$ or $\mathcal{T}_p$.
	Every member of $\mathcal{H}$ is algebraic as an imaginary over $\emptyset$.
	We may therefore use the standard fact that an infinite indiscernible sequence remains indiscernible over $\operatorname{acl}^{\mathrm{eq}}(\emptyset)$.

	Fix $H\in\mathcal{H}$, an integer tuple $\overline{m}$, a nonzero integer $m$, and an integer $k$, where either $k=0$ or $G/H$ is discrete.
	Put $c_i=\overline{m}\cdot\overline{a}_i$ and consider
	\[
		\operatorname{sgn}
		\left(
			(c_i-mb)/H-k_H
		\right).
	\]
	The sequence $(c_i/H)_{i\in I}$ is either constant or strictly monotone.
	Hence, its signs relative to the fixed element $mb/H+k_H$ are constant, or change at a single cut of $I$, with at most one index at which equality holds.

	We claim that two cuts arising in this way cannot have two indices of $I$ strictly between them.
	Suppose toward a contradiction that two configurations determine cuts $\gamma<\delta$, and choose
	\[
		h<\gamma<r<s<\delta<l.
	\]
	Let their convex subgroups be $H$ and $K$.
	Set $L=\max\{H,K\}$ and project both configurations to $G/L$.
	A crossing either remains a crossing at the same cut after projection, or its difference from the relevant fixed element becomes identically zero in $G/L$.
	To justify this, the projection of a monotone indiscernible sequence is again constant or strictly monotone.
	If the projected difference is nonzero somewhere on each side of the original cut, order preservation forces it to have the corresponding negative and positive signs, so its cut is unchanged.
	Otherwise it is zero at two distinct indices; monotonicity then makes it identically zero.
	A nonzero constant is impossible, since its sign would prevent the original difference from changing sign.

	First suppose that both crossings survive in $G/L$.
	After replacing a configuration by its negative if necessary, we may take the coefficient of $b$ to be positive.
	In the divisible hull of $G/L$, write the two configurations as monotone sequences $(u_i)$ and $(v_i)$ crossing the same element $b/L$ at $\gamma$ and $\delta$, respectively; the constants $k_H$ and $k_K$ have been absorbed into $u_i$ and $v_i$.
	This is only shorthand: after clearing positive denominators, every comparison below is a formula over the algebraic imaginary $L$, so indiscernibility applies.

	If $u_i$ and $v_i$ have the same orientation, use the indices chosen above.
	If both are increasing, then
	\[
		u_r>v_s
		\quad\text{but}\quad
		u_h<v_l;
	\]
	if both are decreasing, the two inequalities are reversed.
	This contradicts indiscernibility, since both $(r,s)$ and $(h,l)$ are increasing pairs.
	If the two sequences have opposite orientations, then the truth of $u_i<v_i$ differs on the two sides of $\gamma$ and $\delta$, again contradicting indiscernibility.

	It remains to consider the case where, say, the first crossing vanishes after projection.
	Then $H<L=K$, and hence the second crossing survives.
	Moreover,
	\[
		c_i-mb\in L
	\]
	for every $i$.
	Here $k_H$ maps to zero in $G/L$: if $G/H$ is discrete, every nonzero convex subgroup of $G/H$ contains its least positive element.
	For the second configuration, with $d_i=\overline{n}\cdot\overline{a}_i$ and positive coefficient $m'$, we have in $G/L$
	\[
		m(d_i-m'b-k_K)
		=
		md_i-m'c_j-m(k_K)
	\]
	for every $j$.
	Choosing $j<i$, the sign of the expression on the right is independent of the increasing pair $(j,i)$ by indiscernibility.
	Thus, the second configuration cannot cross at $\delta$, a contradiction.
	The other case is symmetric.
	This proves the claim.

	Since $I$ is dense, the claim implies that all the crossing cuts either coincide, or are the two cuts immediately adjacent to a single index $q$.
	In the latter case put $J=\{q\}$.
	In the former case cut at the common cut, placing its possible endpoint in $J$; if there are no crossings, choose any $q\in I$ and put $J=\{q\}$.
	Let $I^-$ and $I^+$ be the resulting initial and final parts.
	They have no endpoints.
	The quotient signs considered above are constant on each of $I^-$ and $I^+$.
	Taking $H=\{0\}$ and $k=0$ gives (1).
	For (2), note that, because $\mathcal{S}_p$ is finite, $\mathfrak{t}_p(x)$ is determined by the truth values of $x\in H$ for $H\in\mathcal{S}_p$, and these are determined by the signs of $x/H$.
	Finally, by \cite[Lemma 2.21(1)]{Okura2026DisOrd}, a witness to $x=_{\bullet}k_{\bullet}$ is $\mathfrak{t}_2(x)$.
	Thus (2), together with constancy of the sign of $x/H-k_H$, gives (3).
\end{proof}

We now control all contributions from the $\mathfrak{s}_{p^r}$-side.
For fixed $p$ and $H\in\mathcal{S}_p^\infty$, call $i_0\in I$ a \emph{$(p,H)$-exception} if there are an integer tuple $\overline{m}$, a nonzero integer $m$, an integer $r\geq 1$, and an integer $k$ such that either $k=0$ or $G/H$ is discrete, and
\[
	(\overline{m}\cdot\overline{a}_i-mb)/H
	\equiv_{p^r}
	k_H
	\quad\Longleftrightarrow\quad
	i=i_0.
\]

\begin{lem}[Uniqueness of the congruence exception]\label{lem:UniqueCongruenceException}
	For each $p$ and $H\in\mathcal{S}_p^\infty$, there is at most one $(p,H)$-exception.
\end{lem}

\begin{proof}
	Suppose that $i_0$ and $i_1$ are exceptions, witnessed respectively by
	\[
		(c_i,m,r,k)
		\quad\text{and}\quad
		(d_i,m',r',l),
	\]
	where $c_i$ and $d_i$ are fixed integer-linear terms in $\overline{a}_i$.
	If $p^r\mid m$, then the first congruence is independent of $b$.
	It cannot hold at exactly one index of an indiscernible sequence over the algebraic imaginary $H$.
	Thus $v_p(m)<r$, and similarly $v_p(m')<r'$.

	Let $R\geq r,r'$.
	Multiplying the first congruence by $p^{R-r}$ replaces its modulus by $p^R$ without changing its set of solutions.
	Indeed, purity and torsion-freeness of $G/H$ give
	\[
		p^{R-r}z\in H+p^RG
		\quad\Longleftrightarrow\quad
		z\in H+p^rG.
	\]
	Do the same with the second congruence.
	Multiplication by an integer prime to $p$ preserves the entire solution set.
	Using an inverse for the prime-to-$p$ part of each coefficient modulo $p^R$, we may therefore write the two exceptional congruences in the form
	\begin{align*}
		C_{i_0}/H
		&\equiv_{p^R}
		p^e b/H+\kappa_H,\\
		D_{i_1}/H
		&\equiv_{p^R}
		p^f b/H+\lambda_H,
	\end{align*}
	where $0\leq e,f<R$, and each congruence holds at exactly its displayed index.

	Assume first that $i_0<i_1$.
	If $e\leq f$, eliminating $b$ gives
	\[
		(D_{i_1}-p^{f-e}C_{i_0})/H
		\equiv_{p^R}
		\lambda_H-p^{f-e}\kappa_H.
	\]
	By indiscernibility over the algebraic imaginary $H$, the same relation holds with $i_1$ replaced by any $j>i_0$.
	Subtracting yields
	\[
		D_j-D_{i_1}\in H+p^RG.
	\]
	Hence the second exceptional congruence also holds at $j$, a contradiction.
	If $e>f$, eliminate $b$ in the other direction and replace the increasing pair $(i_0,i_1)$ by $(h,i_1)$ for some $h<i_1$, $h\neq i_0$.
	This gives $C_h-C_{i_0}\in H+p^RG$, contradicting uniqueness of the first congruence.
	The case $i_1<i_0$ is symmetric.
\end{proof}

\begin{lem}[Control of the $\mathfrak{s}$-data]\label{lem:ControlOfSData}
	For every pair $(p,H)$ with $H\in\mathcal{S}_p^\infty$, choose its exceptional index if one exists, and let $E$ be the set of indices so obtained.
	Then $|E|\leq N$.
	If $C\subseteq I$ is convex and $C\cap E=\emptyset$, then for any $i,j\in C$, every integer tuple $\overline{m}$, every nonzero integer $m$, every prime $p$, every $r\geq 1$, and every $1\leq k<p^r$, we have
	\begin{align*}
		\mathfrak{s}_{p^r}
		(\overline{m}\cdot\overline{a}_i-mb)
		&=
		\mathfrak{s}_{p^r}
		(\overline{m}\cdot\overline{a}_j-mb),\\
		\overline{m}\cdot\overline{a}_i-mb
		\equiv_{p^r,\bullet}k_{\bullet}
		&\Longleftrightarrow
		\overline{m}\cdot\overline{a}_j-mb
		\equiv_{p^r,\bullet}k_{\bullet}.
	\end{align*}
\end{lem}

\begin{proof}
	The estimate for $|E|$ follows from \autoref{lem:UniqueCongruenceException}.
	Fix
	\[
		x_i=\overline{m}\cdot\overline{a}_i-mb
	\]
	and $p^r$.
	By indiscernibility and finiteness of $\mathcal{S}_p$, there is a fixed value
	\[
		K
		=
		\mathfrak{s}_{p^r}(x_j-x_i)
	\]
	for all $i<j$.
	More precisely, constancy follows because the sequence is indiscernible over the algebraic elements of the finite auxiliary sort.
	The isosceles-triangle property shows that either
	$(\mathfrak{s}_{p^r}(x_i))_{i\in I}$ is constant, or there is a unique $i_0$ such that
	\[
		\mathfrak{s}_{p^r}(x_{i_0})<K
		\quad\text{and}\quad
		\mathfrak{s}_{p^r}(x_i)=K
		\quad(i\neq i_0).
	\]
	Indeed, unequal values at two indices must have maximum $K$.
	Once one value is below $K$, comparison with that index forces every other value to equal $K$; two values below $K$, or any value above $K$, would contradict
	$\mathfrak{s}_{p^r}(x_j-x_i)=K$.
	In the second case, the images of the $x_i$, for $i\neq i_0$, in
	$R_K^{\mathfrak{s}_{p^r}}$ are pairwise distinct.
	Thus this rib is infinite, and \autoref{lem:HigherPrimePowerRibs} gives
	$K\in\mathcal{S}_p^\infty$.
	Moreover,
	\[
		x_i\in K+p^rG
		\quad\Longleftrightarrow\quad
		i=i_0,
	\]
	so $i_0$ is the $(p,K)$-exception with $k=0$.
	A convex set disjoint from $E$ therefore sees constant $\mathfrak{s}_{p^r}$-value.

	Now restrict to such a convex set $C$, and let the common value be $H$.
	If $H=\emptyset$, the predicate
	$x_i\equiv_{p^r,\bullet}k_{\bullet}$ is false throughout $C$.
	Suppose $H\neq\emptyset$.
	By \cite[Lemma 2.21(2)]{Okura2026DisOrd}, if the predicate holds, then its witnessing subgroup is precisely
	$\mathfrak{s}_{p^r}(x_i)=H$.
	If $G/H$ is dense, the predicate is therefore false throughout $C$.
	We may thus assume that $G/H$ is discrete.
	If the value of $\mathfrak{s}_{p^r}(x_j-x_i)$ is less than $H$, then all the $x_i$ have the same image modulo $H+p^rG$, and the predicate is constant.
	Otherwise that value is $H$, and the relevant images are pairwise distinct.
	If the $\mathfrak{s}_{p^r}(x_i)$ are constant on all of $I$, this shows directly that at most one index in $I$ can satisfy
	\[
		x_i/H\equiv_{p^r}k_H.
	\]
	If instead there is an exceptional index $i_0$ outside which all the values equal $H$, then $x_{i_0}\in H+p^rG$.
	Since $1\leq k<p^r$, the class $k_H$ is nonzero modulo $p^r$ in the discrete ordered group $G/H$.
	Thus $i_0$ does not satisfy the displayed congruence, and again at most one index in all of $I$ can satisfy it.
	If such an index exists, the same pairwise-distinctness argument makes
	$R_H^{\mathfrak{s}_{p^r}}$ infinite, so $H\in\mathcal{S}_p^\infty$; the unique satisfying index is then the $(p,H)$-exception.
	It does not belong to $C$.
	Thus the predicate is again constant on $C$.
\end{proof}

\begin{thm}\label{thm:UpperBoundFiniteSpines}
	Let $G$ be a nontrivial ordered abelian group with finite spines, and suppose that
	\[
		\sum_{p\in\mathbb{P}}
		\left|\mathcal{S}_p^\infty\right|
	\]
	is finite.
	Then
	\[
		\operatorname{dp\text{-}rk}(G)
		\leq
		1+
		\sum_{p\in\mathbb{P}}
		\left|\mathcal{S}_p^\infty\right|.
	\]
\end{thm}

\begin{proof}
	We work in a sufficiently saturated monster model.
	Let $I\models\mathrm{DLO}$, let $(\overline{a}_i)_{i\in I}$ be indiscernible, and let $b\in G$.
	Apply \autoref{lem:CommonOrderCut} and \autoref{lem:ControlOfSData}.
	Cut $I$ at the common order cut and at the at most $N$ points in $E$.
	If there are fewer than $N+1$ separators, insert arbitrary additional cuts or singleton points.
	We thereby obtain a decomposition
	\[
		I_0<J_0<I_1<\dots<J_N<I_{N+1},
	\]
	where every $I_l$ has no endpoints and every $J_l$ has at most one element.

	Fix $l$ and $i,j\in I_l$.
	Let $M_i$ be the subgroup generated by the entries of $\overline{a}_i$, and define
	\[
		f_{ij}:\langle M_i,b\rangle
		\longrightarrow
		\langle M_j,b\rangle
	\]
	by
	\[
		f_{ij}
		\left(
			\overline{m}\cdot\overline{a}_i+mb
		\right)
		=
		\overline{m}\cdot\overline{a}_j+mb.
	\]
	The case $m=0$ in the following verifications follows directly from indiscernibility over $\operatorname{acl}^{\mathrm{eq}}(\emptyset)$; in particular, the actual values in the finite auxiliary sorts, and not merely their types, agree.
	For $m\neq0$, \autoref{lem:CommonOrderCut}(1) shows that the sign of every element is preserved.
	If two displayed representations of an element of $\langle M_i,b\rangle$ are equal, subtracting them gives an integer-linear expression of sign zero.
	Its corresponding expression at $j$ also has sign zero, so $f_{ij}$ is well-defined.
	The same argument for nonzero signs gives injectivity and order preservation.
	Thus $f_{ij}$ is an ordered group embedding.
	The same lemma, parts (2) and (3), shows that $f_{ij}$ preserves every $\mathfrak{t}_p$ and every predicate $x=_{\bullet}k_{\bullet}$.
	By \autoref{lem:ControlOfSData}, it preserves every $\mathfrak{s}_{p^r}$ and every predicate $x\equiv_{p^r,\bullet}k_{\bullet}$.
	Therefore, \autoref{prop:FiniteSpineElementaryMap} implies that $f_{ij}$ is elementary.
	Since it fixes $b$ and sends $\overline{a}_i$ to $\overline{a}_j$, we obtain
	\[
		\operatorname{tp}(\overline{a}_i/b)
		=
		\operatorname{tp}(\overline{a}_j/b).
	\]

	We have verified the hypothesis of \autoref{fct:DpRankCriterion} with $n=N+1$.
	Hence $\operatorname{dp\text{-}rk}(G)\leq N+1$, as required.
\end{proof}
